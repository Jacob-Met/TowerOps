def tree(text):
    return ast.parse(text)
def without_imports(text):
    value=tree(text)
    value.body=[node for node in value.body if not isinstance(node,(ast.Import,ast.ImportFrom))]
    return ast.dump(value,include_attributes=False)
def imports(text):
    return sorted(ast.dump(node,include_attributes=False) for node in tree(text).body if isinstance(node,(ast.Import,ast.ImportFrom)))
