def run(name, command, cwd, required=False):
    start = time.monotonic()
    with (out / (name + ".stdout.log")).open("xb") as stdout, (out / (name + ".stderr.log")).open("xb") as stderr:
        try:
            result = subprocess.run(command, cwd=cwd, env=env, stdout=stdout, stderr=stderr, timeout=300)
            code = result.returncode
        except subprocess.TimeoutExpired:
            code = 124
    receipt = {"name": name, "command": command, "cwd": str(cwd), "exit_code": code,
               "seconds": round(time.monotonic()-start, 3)}
    records.append(receipt)
    (out / "process-receipt.json").write_text(json.dumps(records, indent=2) + "\n")
    print(json.dumps(receipt), flush=True)
    if code:
        print((out / (name + ".stderr.log")).read_text(errors="replace")[-3000:], flush=True)
        print((out / (name + ".stdout.log")).read_text(errors="replace")[-3000:], flush=True)
        if required:
            raise SystemExit(code)
    return code
def git(*args):
    return subprocess.check_output(["git", *args], cwd=repo, env=env)
def sha(data):
    return hashlib.sha256(data).hexdigest()
def blob(data):
    return hashlib.sha1(b"blob " + str(len(data)).encode() + b"\0" + data).hexdigest()
def entries(tree):
    result = {}
    for row in git("ls-tree", "-rz", "--full-tree", tree).split(b"\0"):
        if row:
            meta, path = row.split(b"\t", 1)
            mode, kind, digest = meta.decode().split()
            result[path.decode()] = {"mode": mode, "type": kind, "git_blob": digest}
    return result
