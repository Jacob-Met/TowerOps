def identity_bytes(data):
    return {"bytes": len(data), "sha256": hashlib.sha256(data).hexdigest(), "git_blob": hashlib.sha1(b"blob " + str(len(data)).encode() + b"\0" + data).hexdigest()}

def identity(path):
    return {**identity_bytes(path.read_bytes()), "mode": "100755" if path.stat().st_mode & stat.S_IXUSR else "100644"}

def write_json(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes((json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode("utf-8"))
    return identity(path)

def copy_exact(source, relative, expected=None):
    source = Path(source)
    if Path(relative).is_absolute() or ".." in Path(relative).parts:
        raise RuntimeError(f"Unsafe publication path {relative}")
    row = identity(source)
    if expected:
        for key in ("bytes", "sha256", "git_blob", "mode"):
            if key in expected and row[key] != expected[key]:
                raise RuntimeError(f"Source mismatch {source} {key}")
    target = OVERLAY / relative
    if target.exists() and identity(target) != row:
        raise RuntimeError(f"Destination collision {relative}")
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(source.read_bytes())
    target.chmod(0o755 if row["mode"] == "100755" else 0o644)
    if identity(target) != row:
        raise RuntimeError(f"Copy mismatch {relative}")
    return {"source_path": str(source), "path": relative, **row}

def tree_entries(tree):
    result = {}
    for line in subprocess.check_output(["git", "ls-tree", "-rz", tree], cwd=CANDIDATE).split(b"\0"):
        if not line:
            continue
        meta, path = line.split(b"\t", 1)
        mode, kind, blob = meta.decode().split()
        result[path.decode()] = {"mode": mode, "type": kind, "git_blob": blob}
    return result

def stage_overlay():
    paths = sorted(str(p.relative_to(OVERLAY)) for p in OVERLAY.rglob("*") if p.is_file())
    for relative in paths:
        source = OVERLAY / relative
        target = CANDIDATE / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, target)
        if identity(target) != identity(source):
            raise RuntimeError(f"Materialization mismatch {relative}")
        if "\n" in relative:
            raise RuntimeError("Unexpected newline in publication filename")
    hashing = subprocess.run(["git", "hash-object", "-w", "--no-filters", "--stdin-paths"], cwd=CANDIDATE, input=("\n".join(paths) + "\n").encode(), capture_output=True, check=True)
    blobs = hashing.stdout.decode().splitlines()
    if len(blobs) != len(paths):
        raise RuntimeError("Hash-object output length mismatch")
    records = []
    for relative, blob in zip(paths, blobs, strict=True):
        pin = identity(OVERLAY / relative)
        if blob != pin["git_blob"]:
            raise RuntimeError(f"Git filter changed content {relative}")
        records.append(f"{pin['mode']} {blob}\t{relative}\0")
    subprocess.run(["git", "update-index", "-z", "--index-info"], cwd=CANDIDATE, input="".join(records).encode(), capture_output=True, check=True)
    return subprocess.check_output(["git", "write-tree"], cwd=CANDIDATE).decode().strip()

def run_ruff(label, tree):
    started = time.time()
    arguments = [RUFF, "check", "."]
    process = subprocess.run(arguments, cwd=CANDIDATE, capture_output=True)
    stdout = ASSEMBLY / f"{label}.stdout.log"
    stderr = ASSEMBLY / f"{label}.stderr.log"
    stdout.write_bytes(process.stdout)
    stderr.write_bytes(process.stderr)
    receipt = {
        "command": arguments, "cwd": str(CANDIDATE), "input_tree": tree,
        "ruff_version": subprocess.check_output([RUFF, "--version"]).decode().strip(),
        "exit_code": process.returncode, "seconds": time.time() - started,
        "stdout": {"source_path": str(stdout), **identity(stdout)},
        "stderr": {"source_path": str(stderr), **identity(stderr)},
        "extra_ignore_flags": [], "new_lint_policy": False,
    }
    write_json(ASSEMBLY / f"{label}-receipt.json", receipt)
    if process.returncode:
        raise RuntimeError(f"{label} exited {process.returncode}; exact raw output retained")
    return receipt
