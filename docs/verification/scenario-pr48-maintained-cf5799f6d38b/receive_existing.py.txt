"""Receive an existing immutable build with its maintained browser receiver."""
from __future__ import annotations

import argparse
import hashlib
import json
import os
import platform
import shutil
import subprocess
import time
from datetime import datetime, timezone
from pathlib import Path

PYTHON = Path("/tmp/towerops-encounter-scenarios-cf5799f6d38b-1_su10pa/playwright-tools/bin/python")
CHROME = Path("/Applications/Google Chrome.app/Contents/MacOS/Google Chrome")
RECEIVER = "tools/check_advisory_options_browser.py"
RECEIVER_BLOB = "9746bd30b3ca2fb2b0a56db45a461c0f2884afaa"


def identity(path: Path) -> dict:
    data = path.read_bytes()
    return {
        "bytes": len(data),
        "sha256": hashlib.sha256(data).hexdigest(),
        "git_blob": hashlib.sha1(b"blob " + str(len(data)).encode() + b"\0" + data).hexdigest(),
        "mode": "100755" if path.stat().st_mode & 0o111 else "100644",
    }


def write_json(path: Path, value: object) -> None:
    path.write_text(json.dumps(value, indent=2) + "\n")


def snapshot(pin: dict) -> dict:
    source_root = Path(pin["source_root"])
    build_root = Path(pin["build_root"])
    source = [{"path": row["path"], **identity(source_root / row["path"])} for row in pin["source_files"]]
    build = [
        {"path": str(path.relative_to(build_root)), **identity(path)}
        for path in sorted(build_root.rglob("*")) if path.is_file()
    ]
    mismatches = []
    for category, actual_rows, expected_rows in (
        ("source", source, pin["source_files"]),
        ("build", build, pin["build_files"]),
    ):
        expected = {row["path"]: row for row in expected_rows}
        actual = {row["path"]: row for row in actual_rows}
        if set(actual) != set(expected):
            mismatches.append({"category": category, "path_set_changed": True})
        for name in expected.keys() & actual.keys():
            changed = {
                key: {"expected": expected[name][key], "actual": actual[name][key]}
                for key in ("bytes", "sha256", "git_blob", "mode")
                if key in expected[name] and expected[name][key] != actual[name][key]
            }
            if changed:
                mismatches.append({"category": category, "path": name, "changed": changed})
    return {"source_files": source, "build_files": build, "mismatches": mismatches}


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--pin", type=Path, required=True)
    parser.add_argument("--expected-tree", required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=False)
    pin_bytes = args.pin.read_bytes()
    pin = json.loads(pin_bytes)
    if pin["source_tree"] != args.expected_tree:
        raise RuntimeError("Unexpected source tree supplied")
    (args.output / "supplied-source-build-pin.json").write_bytes(pin_bytes)
    before = snapshot(pin)
    write_json(args.output / "source-build-before.json", before)
    if before["mismatches"]:
        raise RuntimeError("Source/build differs from the supplied pin before receiving")
    source_root = Path(pin["source_root"])
    receiver = source_root / RECEIVER
    receiver_identity = identity(receiver)
    if receiver_identity["git_blob"] != RECEIVER_BLOB:
        raise RuntimeError("Maintained receiver bytes changed")
    env = os.environ.copy()
    env.update({"PYTHONDONTWRITEBYTECODE": "1", "PYTHONUNBUFFERED": "1", "PYTHONOPTIMIZE": "0"})
    runtime_tmp = args.output / "runtime-tmp"
    runtime_tmp.mkdir()
    env["TMPDIR"] = str(runtime_tmp)
    runtime_code = (
        "import json,sys,importlib.metadata;"
        "print(json.dumps({'python':sys.version,'executable':sys.executable,"
        "'optimization':sys.flags.optimize,'playwright':importlib.metadata.version('playwright')}))"
    )
    runtime_process = subprocess.run(
        [str(PYTHON), "-B", "-c", runtime_code], env=env,
        capture_output=True, text=True, check=True,
    )
    runtime = json.loads(runtime_process.stdout)
    if runtime["optimization"] != 0:
        raise RuntimeError("Receiver assertions must remain enabled")
    write_json(args.output / "runtime.json", {
        **runtime, "platform": platform.platform(), "chrome_path": str(CHROME),
        "disk_free_before": shutil.disk_usage(args.output).free,
    })
    command = [
        str(PYTHON), "-B", str(receiver), "--chrome", str(CHROME),
        "--output-dir", str(args.output / "browser"),
    ]
    started_utc = datetime.now(timezone.utc).isoformat()
    started = time.monotonic()
    code = None
    failure = None
    try:
        with (args.output / "receiver.stdout.log").open("wb") as stdout:
            with (args.output / "receiver.stderr.log").open("wb") as stderr:
                process = subprocess.run(
                    command, cwd=source_root, env=env, stdout=stdout, stderr=stderr,
                    check=False, timeout=420,
                )
                code = process.returncode
    except Exception as error:
        failure = type(error).__name__ + ": " + str(error)
    elapsed = time.monotonic() - started
    after = snapshot(pin)
    write_json(args.output / "source-build-after.json", after)
    browser_receipt = args.output / "browser/browser-receipt.json"
    browser = json.loads(browser_receipt.read_text()) if browser_receipt.is_file() else None
    receipt = {
        "schema": "towerops-maintained-readonly-receiving-v1",
        "source_tree": pin["source_tree"], "base_commit": pin["base_commit"],
        "source_root": str(source_root), "build_root": pin["build_root"],
        "supplied_pin": {"path": str(args.pin), **identity(args.pin)},
        "receiver": {"path": str(receiver), **receiver_identity},
        "command": command, "started_utc": started_utc, "elapsed_seconds": elapsed,
        "exit_code": code, "driver_error": failure,
        "source_count": len(before["source_files"]), "build_count": len(before["build_files"]),
        "source_build_unchanged": before == after,
        "source_build_mismatches_after": after["mismatches"],
        "passed": browser["passed"] if browser else [],
        "browser_version": browser["browser"] if browser else None,
        "page_errors": browser["page_errors"] if browser else None,
        "external_requests": browser["external_requests"] if browser else None,
        "disk_free_after": shutil.disk_usage(args.output).free,
    }
    write_json(args.output / "receiving-receipt.json", receipt)
    print(json.dumps(receipt, indent=2))
    return 0 if (
        code == 0 and failure is None and before == after
        and not after["mismatches"] and browser and len(browser["passed"]) == 10
    ) else 1


if __name__ == "__main__":
    raise SystemExit(main())
